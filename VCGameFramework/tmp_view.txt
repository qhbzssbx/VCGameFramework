using System;
using System.Collections.Generic;
using UnityEngine;
using Cysharp.Threading.Tasks;
using Game.Infrastructure.Resource.Core;

namespace Game.UI.Core
{
    /// <summary>
    /// 基于YooAsset的UI资源加载器实现
    /// </summary>
    public class YooAssetUIResourceLoader : IUIResourceLoader
    {
        private readonly ResourceLoader _resourceLoader;
        private readonly Dictionary<string, UnityEngine.Object> _assetCache;
        private readonly Dictionary<Type, string> _typePrefabCache;
        private bool _disposed = false;
        
        public YooAssetUIResourceLoader()
        {
            _resourceLoader = new ResourceLoader();
            _assetCache = new Dictionary<string, UnityEngine.Object>();
            _typePrefabCache = new Dictionary<Type, string>();
        }
        
        /// <summary>
        /// 异步加载UI预制体
        /// </summary>
        public async UniTask<GameObject> LoadUIPrefabAsync(string assetKey)
        {
            if (_disposed)
            {
                throw new ObjectDisposedException(nameof(YooAssetUIResourceLoader));
            }
            
            // 检查缓存
            if (_assetCache.TryGetValue(assetKey, out var cachedAsset) && cachedAsset is GameObject cachedPrefab)
            {
                return cachedPrefab;
            }
            
            try
            {
                // 加载资源
                var handle = await _resourceLoader.LoadAssetAsync<GameObject>($"Assets/AssetRaw/UI/{assetKey}.prefab");
                if (handle.IsValid && handle.AssetObject != null)
                {
                    var prefab = handle.AssetObject as GameObject;
                    
                    // 缓存资源
                    _assetCache[assetKey] = prefab;
                    
                    return prefab;
                }
                
                Debug.LogError($"UI预制体加载失败: {assetKey}");
                return null;
            }
            catch (Exception ex)
            {
                Debug.LogError($"UI预制体加载异常: {assetKey}, Error: {ex.Message}");
                return null;
            }
        }
        
        /// <summary>
        /// 加载UI纹理
        /// </summary>
        public async UniTask<Texture2D> LoadUITextureAsync(string assetKey)
        {
            if (_disposed)
            {
                throw new ObjectDisposedException(nameof(YooAssetUIResourceLoader));
            }
            
            // 检查缓存
            if (_assetCache.TryGetValue(assetKey, out var cachedAsset) && cachedAsset is Texture2D cachedTexture)
            {
                return cachedTexture;
            }
            
            try
            {
                var handle = await _resourceLoader.LoadAssetAsync<Texture2D>(assetKey);
                if (handle.IsValid && handle.AssetObject != null)
                {
                    var texture = handle.AssetObject as Texture2D;
                    _assetCache[assetKey] = texture;
                    return texture;
                }
                
                Debug.LogError($"UI纹理加载失败: {assetKey}");
                return null;
            }
            catch (Exception ex)
            {
                Debug.LogError($"UI纹理加载异常: {assetKey}, Error: {ex.Message}");
                return null;
            }
        }
        
        /// <summary>
        /// 加载UI精灵
        /// </summary>
        public async UniTask<Sprite> LoadUISpriteAsync(string assetKey)
        {
            if (_disposed)
            {
                throw new ObjectDisposedException(nameof(YooAssetUIResourceLoader));
            }
            
            // 检查缓存
            if (_assetCache.TryGetValue(assetKey, out var cachedAsset) && cachedAsset is Sprite cachedSprite)
            {
                return cachedSprite;
            }
            
            try
            {
                var handle = await _resourceLoader.LoadAssetAsync<Sprite>(assetKey);
                if (handle.IsValid && handle.AssetObject != null)
                {
                    var sprite = handle.AssetObject as Sprite;
                    _assetCache[assetKey] = sprite;
                    return sprite;
                }
                
                Debug.LogError($"UI精灵加载失败: {assetKey}");
                return null;
            }
            catch (Exception ex)
            {
                Debug.LogError($"UI精灵加载异常: {assetKey}, Error: {ex.Message}");
                return null;
            }
        }
        
        /// <summary>
        /// 预加载UI资源
        /// </summary>
        public async UniTask PreloadUIAssetsAsync(params string[] assetKeys)
        {
            if (_disposed)
            {
                throw new ObjectDisposedException(nameof(YooAssetUIResourceLoader));
            }
            
            var loadTasks = new List<UniTask>();
            
            foreach (var assetKey in assetKeys)
            {
                if (!_assetCache.ContainsKey(assetKey))
                {
                    // 根据资源键名推断资源类型
                    if (assetKey.Contains("Prefab") || assetKey.Contains("UI"))
                    {
                        loadTasks.Add(LoadUIPrefabAsync(assetKey).AsUniTask());
                    }
                    else if (assetKey.Contains("Texture") || assetKey.Contains("Image"))
                    {
                        loadTasks.Add(LoadUITextureAsync(assetKey).AsUniTask());
                    }
                    else if (assetKey.Contains("Sprite") || assetKey.Contains("Icon"))
                    {
                        loadTasks.Add(LoadUISpriteAsync(assetKey).AsUniTask());
                    }
                    else
                    {
                        // 默认尝试作为预制体加载
                        loadTasks.Add(LoadUIPrefabAsync(assetKey).AsUniTask());
                    }
                }
            }
            
            if (loadTasks.Count > 0)
            {
                try
                {
                    await UniTask.WhenAll(loadTasks);
                    Debug.Log($"UI资源预加载完成，共加载 {loadTasks.Count} 个资源");
                }
                catch (Exception ex)
                {
                    Debug.LogError($"UI资源预加载失败: {ex.Message}");
                }
            }
        }
        
        /// <summary>
        /// 释放UI资源
        /// </summary>
        public void ReleaseUIAsset(string assetKey)
        {
            if (_assetCache.TryGetValue(assetKey, out var asset))
            {
                _assetCache.Remove(assetKey);
                
                // 如果是UnityEngine.Object，可以考虑调用Resources.UnloadAsset
                // 但YooAsset有自己的资源管理，这里主要是清理缓存
                Debug.Log($"UI资源已从缓存中移除: {assetKey}");
            }
        }
        
        /// <summary>
        /// 清理所有UI资源缓存
        /// </summary>
        public void ClearCache()
        {
            var count = _assetCache.Count;
            _assetCache.Clear();
            _typePrefabCache.Clear();
            
            Debug.Log($"UI资源缓存已清理，共清理 {count} 个资源");
        }
        
        /// <summary>
        /// 注册UI类型与预制体的映射关系
        /// </summary>
        /// <typeparam name="T">UI面板类型</typeparam>
        /// <param name="assetKey">对应的预制体资源键名</param>
        public void RegisterUIPrefab<T>(string assetKey) where T : UIPanel
        {
            _typePrefabCache[typeof(T)] = assetKey;
        }
        
        /// <summary>
        /// 根据UI类型获取预制体资源键名
        /// </summary>
        /// <typeparam name="T">UI面板类型</typeparam>
        /// <returns>预制体资源键名，如果未注册则返回类型名</returns>
        public string GetUIPrefabKey<T>() where T : UIPanel
        {
            if (_typePrefabCache.TryGetValue(typeof(T), out var assetKey))
            {
                return assetKey;
            }
            
            // 如果没有注册，使用类型名作为默认键名
            return typeof(T).Name;
        }
        
        /// <summary>
        /// 释放资源
        /// </summary>
        public void Dispose()
        {
            if (_disposed) return;
            
            try
            {
                ClearCache();
                _resourceLoader?.Dispose();
                _disposed = true;
                
                Debug.Log("YooAssetUIResourceLoader 已释放");
            }
            catch (Exception ex)
            {
                Debug.LogError($"YooAssetUIResourceLoader 释放时发生错误: {ex.Message}");
            }
        }
    }
}
